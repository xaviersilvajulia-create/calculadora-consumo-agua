Etapas do Projeto & Script em Python

ETAPA 1

Investigação do Problema

  • Problema: Consumo excessivo e desperdício de água potável por falta de visibilidade sobre os gastos diários e ocultos.
  • Onde ocorre: Residências dos alunos, banheiros, pátio e cozinha da escola.
  • Afetados: Comunidade escolar, orçamento das famílias e mananciais locais sujeitos à estiagem.
  • Papel da Tecnologia: Converter hábitos em indicadores visuais claros, simular o Dia Zero e incentivar ações coletivas.
ETAPA 2

Planejamento da Solução

  • Objetivo: Calcular o consumo diário, mensal e anual, comparar com a meta da ONU (110 L/dia) e projetar economias.
  • Entradas: Banhos, descargas, torneiras, máquina de lavar, mangueira e hábitos de consumo invisível.
  • Processamento: Multiplicação por vazões médias em litros, cálculo de excedente vital (50 L/dia) e simulação de reservatórios.
  • Saídas: Painel interativo responsivo na web e programa estruturado em Python.

# Etapa 3 - AquaImpacto: Calculadora de Consumo de Água e Consciência Ambiental
# Vazões médias utilizadas (litros):
# Chuveiro: 9 L/min | Descarga: 10 L/uso | Torneira higiene: 6 L/min
# Torneira louça: 8 L/min | Máquina lavar: 135 L/ciclo | Mangueira: 18 L/min


def ler_numero(mensagem: str) -> float:
    """Lê e valida um número não negativo informado pelo usuário."""
    while True:
        try:
            valor = float(input(mensagem).replace(",", "."))
            if valor < 0:
                print("Erro: Digite um valor maior ou igual a zero.")
            else:
                return valor
        except ValueError:
            print("Entrada inválida. Digite apenas números (ex: 10 ou 5.5).")


def executar_calculadora() -> None:
    print("\n============================================================")
    print("   AQUAIMPACTO - MONITOR DE CONSUMO E CONSCIÊNCIA HÍDRICA")
    print("============================================================")

    qtd_banhos = ler_numero("1. Quantos banhos você toma por dia? ")
    tempo_banho = ler_numero("2. Duração média de cada banho (minutos): ")
    descargas = ler_numero("3. Quantas descargas aciona por dia? ")
    higiene = ler_numero("4. Minutos/dia com torneira aberta (dentes/mãos): ")
    louca = ler_numero("5. Minutos/dia lavando louça com torneira aberta: ")
    maquina_sem = ler_numero("6. Vezes por SEMANA que usa máquina de lavar: ")
    mangueira_sem = ler_numero("7. Minutos por SEMANA usando mangueira: ")

    atividades = {
        "Banho": qtd_banhos * tempo_banho * 9.0,
        "Descarga sanitária": descargas * 10.0,
        "Torneira (dentes/mãos)": higiene * 6.0,
        "Lavar louça": louca * 8.0,
        "Máquina de lavar": (maquina_sem * 135.0) / 7.0,
        "Uso de mangueira": (mangueira_sem * 18.0) / 7.0,
    }

    consumo_diario = sum(atividades.values())
    consumo_mensal = consumo_diario * 30
    consumo_anual = consumo_diario * 365
    excesso_diario = max(0.0, consumo_diario - 110.0)
    pessoas_atendidas = int(excesso_diario // 50)
    caixas_agua_ano = consumo_anual / 1000.0
    custo_anual = caixas_agua_ano * 8.50
    maior_vilao = max(atividades, key=atividades.get)

    # Simulação Dia Zero (Escola: 800 pessoas | Reservatório útil: 450.000 L)
    dias_escola = int(450000 // max(1.0, consumo_diario * 800))

    print("\n--- DIAGNÓSTICO DE CONSUMO ---")
    for nome, litros in atividades.items():
        print(f"- {nome:<24}: {litros:6.1f} L/dia")

    print("------------------------------------------------------------")
    print(f"CONSUMO DIÁRIO  : {consumo_diario:.1f} litros (Meta ONU: 110 L/dia)")
    print(f"CONSUMO MENSAL  : {consumo_mensal:.1f} litros ({consumo_mensal / 1000:.2f} m³)")
    print(f"CONSUMO ANUAL   : {consumo_anual:,.0f} litros ({caixas_agua_ano:.1f} caixas de 1.000L)")
    print(f"CUSTO ESTIMADO  : R$ {custo_anual:.2f}/ano (em 5 anos: R$ {custo_anual * 5:.2f})")
    print(f"DIA ZERO ESCOLA : {dias_escola} dias (para 800 alunos com tanque de 500 mil L)")
    print("------------------------------------------------------------")

    if consumo_diario <= 110:
        print("STATUS: SUSTENTÁVEL (Dentro da meta recomendada pela ONU).")
    elif consumo_diario <= 180:
        print("STATUS: ALERTA DE EXCESSO (Acima da meta de 110 L/dia).")
    else:
        print("STATUS: CRÍTICO (Alto desperdício hídrico).")

    if pessoas_atendidas > 0:
        print(f"IMPACTO SOCIAL: Seu excedente supriria o mínimo vital de {pessoas_atendidas} pessoa(s)/dia.")

    print(f"MAIOR VILÃO   : {maior_vilao} ({atividades[maior_vilao]:.1f} L/dia)")
    print("============================================================\n")


if __name__ == "__main__":
    while True:
        executar_calculadora()
        repetir = input("Deseja fazer outra simulação? (S/N): ").strip().upper()
        if repetir != "S":
            print("Programa encerrado. Preserve cada gota!")
            break
